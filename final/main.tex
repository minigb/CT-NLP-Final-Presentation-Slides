\begin{frame}{Audio Tokens Do Not Reliably Encode Musical Structure}

  {\small
  \begin{tabular}{p{5.2cm}p{6.5cm}}
    \toprule
    \textbf{What audio tokens can encode} & \textbf{What they miss} \\
    \midrule
    Timbre, loudness, texture & Key signature, chord progression \\
    General instrument colour & Stem-level instrument identity \\
    Rough tempo feel          & Exact meter and beat phase \\
    High-level scene          & Phrase boundaries, section structure \\
    \bottomrule
  \end{tabular}}

  \smallskip
  \begin{tcolorbox}[key]
    The central gap is structural: audio tokens can capture sound texture while
    missing the discrete musical relations needed for reasoning and control.
  \end{tcolorbox}
\end{frame}

    \node[box,  right=1.6cm of rvq] (dec) {Decoder\\{\footnotesize symm.\ 1D CNN}};

    \draw[arr] (enc) -- (rvq);
    \draw[arr] (rvq) -- (dec);

    \node[lbl, above=0.15cm of enc] {Input: $53 \times 300$ frames};
    \node[lbl, above=0.15cm of rvq] {$4 \times 30$ discrete tokens};
    \node[lbl, above=0.15cm of dec] {Recon: $53 \times 300$ frames};
    \node[lbl, below=0.15cm of enc] {var.\ len $\to$ \textbf{30 fr}};

  \end{center}

  \vspace{0.2cm}
  {\small
  \begin{tabular}{p{3.0cm}p{9.2cm}}
    \midrule
    Encoder & 2-layer 1D CNN + temporal pooling; maps variable-length input to 30 frames \\
    Codebook & 4 RVQ stages, 512 codes each; EMA updates + codebook restart \\
    Training loss & BCE for binary dims + MSE for continuous dims; 42{,}000 steps \\
    \midrule
  \end{tabular}}

  \vspace{2pt}
  {\footnotesize\color{Muted}%
   $4{\times}30 = 120$ tokens per window. Same RVQ format as $C_a$ — stream-native positional encoding.}
\end{frame}

\begin{frame}{Current Scope and Next Steps}

  {\small
